    The core test helper for SYNC-06 tests:
    ```python
    def _compute_pixel_rms_for_trajectory(
        self,
        obMotor: SimulatedMotorInterface,
        obClock: FakeClock,
        flTargetAngleDegrees: float,
        iDurationFrames: int,
        flFrameIntervalSeconds: float = 1.0/30.0,
        flSimStepSeconds: float = 0.001
    ) -> float:
        """
        Run a trajectory, compute RMS pixel error between interpolated and true angles.
        """
        flFovDegrees = 6.77
        iWidthPixels = 1280
        flDegreesPerPixel = flFovDegrees / iWidthPixels

        vPixelErrors = []
        for iFrame in range(iDurationFrames):
            # Advance simulation in fine steps
            iStepsPerFrame = int(flFrameIntervalSeconds / flSimStepSeconds)
            for _ in range(iStepsPerFrame):
                obMotor.advance_simulation(flSimStepSeconds)
                obClock.advance_time_seconds(flSimStepSeconds)

            dFrameTime = obClock.get_time_seconds()
            flInterpolated = obMotor.get_estimated_motor_angle_degrees(dFrameTime)

            # True angle from simulation state
            obState = obMotor.get_latest_motor_state()
            flTrue = obState.flMotorAngleDegrees

            flAngleError = abs(flInterpolated - flTrue)
            flPixelError = flAngleError / flDegreesPerPixel
            vPixelErrors.append(flPixelError)

        return float(numpy.sqrt(numpy.mean(numpy.square(vPixelErrors))))
    ```

    IMPORTANT: The SimulatedMotorInterface must be created with FakeClock so that advance_simulation() records timestamps from the FakeClock (which is also the timestamp source for get_estimated_motor_angle_degrees). This makes the test fully deterministic.
