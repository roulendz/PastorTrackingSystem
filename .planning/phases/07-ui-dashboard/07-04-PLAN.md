       """Tap-and-pass structlog processor for the Phase 7 status panel (UI-04).

       Inserted into the structlog chain by ``configure_logging`` BEFORE
       ``JSONRenderer`` per RESEARCH §5 (the corrective finding vs CONTEXT.md D-16
       wording). The processor:
           1. Inspects ``event_dict.get('level')`` (set earlier by
              ``add_log_level``).
           2. Appends ``(level, event, timestamp)`` triples to a bounded
              ``collections.deque`` when level is warning / error / critical.
           3. Returns ``event_dict`` UNCHANGED — chain continues to
              ``JSONRenderer``.

       Output JSON is unchanged. The deque is read by the Phase 7 status
       panel to display the most recent operator-relevant event.
       """
       from __future__ import annotations

       import collections
       from typing import Any, Final

       EventTriple = tuple[str, str, str]   # (level, event, timestamp-iso)
       EventBuffer = collections.deque[EventTriple]

       _CAPTURED_LEVELS: Final[frozenset[str]] = frozenset({"warning", "error", "critical"})
       EVENT_BUS_MAXLEN: Final[int] = 64    # public — UI deque construction site uses this

       def make_event_bus() -> EventBuffer:
           """Factory: returns a fresh bounded deque. Convenience for callers."""
           return collections.deque(maxlen=EVENT_BUS_MAXLEN)


       class EventBusProcessor:
           """structlog processor: tap WARN/ERROR/CRITICAL into a bounded deque.

           See RESEARCH §5 — corrective insertion point is BEFORE ``JSONRenderer``.
           """

           __slots__ = ("_buf",)

           def __init__(self, buffer: EventBuffer) -> None:
               self._buf = buffer

           def __call__(
               self,
               logger: Any,             # structlog's BoundLogger — opaque here
               method_name: str,
               event_dict: dict[str, Any],
           ) -> dict[str, Any]:
               """Tap-and-pass. Mutates the deque, returns event_dict identity-unchanged."""
               level = event_dict.get("level")
               if level in _CAPTURED_LEVELS:
                   self._buf.appendleft((
                       str(level),
                       str(event_dict.get("event", "")),
                       str(event_dict.get("timestamp", "")),
                   ))
               return event_dict
       ```

           level: str = "INFO",
           event_bus_buffer: "EventBuffer | None" = None,
       ) -> None:
           """Configure structlog + stdlib logging for JSON output.

           CALL ONCE at process startup. NOT idempotent: ``logging.basicConfig``

           ``cache_logger_on_first_use=True`` (below) freezes the filtering

               structlog.contextvars.merge_contextvars,
               structlog.processors.add_log_level,

               wrapper_class=structlog.make_filtering_bound_logger(

               self,
               tag_state: int, tag_motor: int, tag_fps: int,

           ) -> None:

               self._tag_err = tag_err

           def record_frame(self, frame_timestamp_ns: int) -> None:
               """Called from Dashboard render tick on EVERY rendered frame (D-15 FPS source)."""
               self._frame_counter += 1
               self._fps_window.append((frame_timestamp_ns, self._frame_counter))

               """10 Hz tick: push 6 formatted strings to the reserved widgets (D-15)."""
               dpg.set_value(self._tag_state, _format_state_line(snap.state, unsaved_badge))
               dpg.set_value(self._tag_motor, _format_motor(snap.motor_state))
               now_ns = snap.last_frame_ts_ns if snap.last_frame_ts_ns is not None else 0
               dpg.set_value(self._tag_fps, _format_fps(compute_fps(self._fps_window, now_ns)))
               dpg.set_value(self._tag_conf, _format_confidence(snap.last_detection_confidence))
               dpg.set_value(self._tag_lock, _format_id_lock(snap.last_locked_track_id))
               dpg.set_value(self._tag_err, _format_last_error(self._event_bus))
       ```
