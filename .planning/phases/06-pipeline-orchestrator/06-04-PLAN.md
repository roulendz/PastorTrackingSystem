```python
"""Composition helper for pipeline integration tests.

Builds a Pipeline backed by the four in-tree fakes:
    * FakeSerialTransport (Phase 2)
    * FakeVideoSource     (Phase 3)
    * FakePoseEngine      (Phase 4)
    * trajectories.*      (Phase 5)

Caller is responsible for `await pipeline.quit()` in a finally block.

from __future__ import annotations

from collections.abc import Iterable
from typing import Any, Final

import numpy as np

from pastor_tracker.config import Config
from pastor_tracker.control.command_dispatcher import CommandDispatcher
from pastor_tracker.control.pan_controller import PanController

from tests.fixtures.camera_traces import FakeVideoSource, _ScriptedFrame, make_solid_bgr
from tests.fixtures.pose_traces import FakePoseEngine, make_detection

_DEFAULT_FRAME_COUNT: Final[int] = 90  # 3 sec @ 30 fps
_DEFAULT_FRAME_DELAY_SEC: Final[float] = 1.0 / 30.0


def _make_valid_config_dict(**overrides: Any) -> dict[str, Any]:
    """Return a valid Config kwargs dict; spreads overrides on top."""
    base: dict[str, Any] = {
        "arduino_port": "COM_FAKE",
        "obs_camera_name": "OBS Virtual Camera",
        "capture_width": 320,
        "capture_height": 240,
        "capture_fps": 30,
        "yolo_model_path": "yolo11n-pose.pt",
        "motor_angle_min_deg": -90.0,
        "motor_angle_max_deg": 90.0,
    }
    base.update(overrides)
    return base


class _FakeFilterGraph:
    """Minimal pygrabber.FilterGraph stand-in for ObsCamera discovery."""
    def __init__(self, devices: list[str]) -> None:
        self._devices = devices
    def get_input_devices(self) -> list[str]:
        return self._devices


def _default_frame_script(width: int, height: int) -> list[_ScriptedFrame]:
    bgr = make_solid_bgr(width, height, (0, 0, 0))
    return [
        _ScriptedFrame(bgr=bgr, ok=True, delay_sec=_DEFAULT_FRAME_DELAY_SEC)
        for _ in range(_DEFAULT_FRAME_COUNT)
    ]


def _default_detection_script(count: int) -> list[list[Detection]]:
    """One centered subject per frame at 30 fps timestamps."""
    return [
        [make_detection(cx=0.5, cy=0.5, track_id=1, conf=0.9,
                        timestamp_ns=k * 33_333_333)]
        for k in range(count)
    ]


async def _pipeline_with_fakes(
    config_overrides: dict[str, Any] | None = None,
    *,
    detection_script: Iterable[list[Detection]] | None = None,
    frame_script: list[_ScriptedFrame] | None = None,
    pose_engine: PoseEngine | None = None,
    fake_serial: FakeSerialTransport | None = None,
) -> tuple[Pipeline, FakeSerialTransport, FakeVideoSource, PoseEngine]:
    """Build a Pipeline backed by all four fakes. Caller awaits quit()."""
    config = Config(**_make_valid_config_dict(**(config_overrides or {})))
    fake_serial = fake_serial if fake_serial is not None else FakeSerialTransport()
    for line in ARDUINO_TRACE_BOOT_ONLY:
        fake_serial.feed_rx(line)
    motor = ArduinoMotor(fake_serial, config)
    fake_video = FakeVideoSource(
        frame_script or _default_frame_script(config.capture_width, config.capture_height),
        width=config.capture_width,
        height=config.capture_height,
    )
    camera = ObsCamera(
        config,
        video_source_factory=lambda idx, w, h, fps: fake_video,
        filter_graph_factory=lambda: _FakeFilterGraph(["OBS Virtual Camera"]),
    )
    if pose_engine is None:
        pose_engine = FakePoseEngine(
            list(detection_script) if detection_script is not None
            else _default_detection_script(_DEFAULT_FRAME_COUNT)
        )
    detector = PoseDetector(config=config, engine=pose_engine)
    pipeline = Pipeline(
        config,
        camera=camera, motor=motor, detector=detector,
        tracker=SubjectTracker(config),
        analyzer=MotionAnalyzer(config),
        framer=Framer(config),
        controller=PanController(config),
        dispatcher=CommandDispatcher(config),
    )
    return pipeline, fake_serial, fake_video, pose_engine
```

```python
async def _force_state(pipeline: Pipeline, target: _PipelineState) -> None:
    """Drive pipeline into target state via the shortest valid sequence."""
    if target is _PipelineState.STOPPED:
        return  # initial state
    if target is _PipelineState.RUNNING:
        await pipeline.start()
        return
    if target is _PipelineState.PAUSED:
        await pipeline.start()

        await pipeline.e_stop()
        return
    if target is _PipelineState.QUITTING:
        await pipeline.quit()
        return
```
